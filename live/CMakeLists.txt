cmake_minimum_required(VERSION 3.16)
project(BuildQueue LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "BuildQueue targets Linux. Use WSL2 or a Linux container on Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(checksum_queue src/BatchQueue.cpp src/Checksum.cpp src/DeadlockDetector.cpp src/DeadlockDemo.cpp)
target_include_directories(checksum_queue PUBLIC include)
target_link_libraries(checksum_queue PUBLIC Threads::Threads)
target_compile_options(checksum_queue PRIVATE -Wall -Wextra -Wpedantic)
add_executable(buildqueue src/main.cpp)
target_include_directories(buildqueue SYSTEM PRIVATE third_party)
target_link_libraries(buildqueue PRIVATE checksum_queue)
target_compile_options(buildqueue PRIVATE -Wall -Wextra -Wpedantic)

option(BUILDQUEUE_QT "Build the Qt Widgets desktop app" OFF)
if(BUILDQUEUE_QT)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    add_executable(buildqueue-qt desktop/main.cpp)
    target_link_libraries(buildqueue-qt PRIVATE checksum_queue Qt6::Widgets)
    # The shared queue API uses "slots" as a field name.
    target_compile_definitions(buildqueue-qt PRIVATE QT_NO_KEYWORDS)
    target_compile_options(buildqueue-qt PRIVATE -Wall -Wextra -Wpedantic)
    file(COPY demo-files DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(deadlock_test tests/test_deadlock.cpp)
    target_link_libraries(deadlock_test PRIVATE checksum_queue)
    add_test(NAME lock_cycle COMMAND deadlock_test "${CMAKE_CURRENT_SOURCE_DIR}/demo-files/deadlock")
    set_tests_properties(lock_cycle PROPERTIES TIMEOUT 30)
    add_executable(live_tests tests/test_live.cpp)
    target_link_libraries(live_tests PRIVATE checksum_queue)
    add_test(NAME live_workers COMMAND live_tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME live_http COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_http.py
            $<TARGET_FILE:buildqueue> ${CMAKE_CURRENT_SOURCE_DIR}/web)
    endif()
endif()
